class PaymentMethod(ABC):

    @abstractmethod
    def pay(self, amount):
        pass

class UPIPayment(PaymentMethod):
    def pay(self, amount):
        return f"₹{amount} paid through UPI"

class CardPayment(PaymentMethod):
    def pay(self, amount):
        return f"₹{amount} paid through card"

class NetbankingPayment(PaymentMethod):
    def pay(self, amount):
        return f"₹{amount} paid through Netbanking"        

def checkout(method: PaymentMethod, amount):
    return method.pay(amount)
